// Global değişkenler let walletAddresses = []; let allPositions = []; let currentFilter = 'all'; const API_URL = 'https://api.hyperliquid.xyz/info'; // Whale kategorileri const WHALE_CATEGORIES = { small: { min: 100000, max: 500000, name: 'Small Whale', emoji: '🐟' }, whale: { min: 500000, max: 1000000, name: 'Whale', emoji: '🐋' }, tidal: { min: 1000000, max: 50000000, name: 'Tidal Whale', emoji: '🌊' }, mega: { min: 50000000, max: Infinity, name: 'Mega Whale', emoji: '🦈' } }; // LocalStorage'dan cüzdanları yükle function loadWalletsFromStorage() { const stored = localStorage.getItem('hyperliquid_wallets'); if (stored) { try { walletAddresses = JSON.parse(stored); updateWalletList(); } catch (e) { console.error('Storage hatası:', e); } } } // LocalStorage'a cüzdanları kaydet function saveWalletsToStorage() { localStorage.setItem('hyperliquid_wallets', JSON.stringify(walletAddresses)); } // Whale kategorisini belirle function getWhaleCategory(value) { if (value >= WHALE_CATEGORIES.mega.min) return 'mega'; if (value >= WHALE_CATEGORIES.tidal.min) return 'tidal'; if (value >= WHALE_CATEGORIES.whale.min) return 'whale'; if (value >= WHALE_CATEGORIES.small.min) return 'small'; return null; } // Cüzdan ekle function addWallet() { const input = document.getElementById('walletInput'); let address = input.value.trim(); if (!address) { alert('Lütfen bir cüzdan adresi girin'); return; } if (!address.startsWith('0x')) { address = '0x' + address; } if (address.length !== 42) { alert('Geçersiz Ethereum adresi (42 karakter olmalı)'); return; } if (walletAddresses.includes(address.toLowerCase())) { alert('Bu adres zaten ekli'); return; } walletAddresses.push(address.toLowerCase()); saveWalletsToStorage(); updateWalletList(); input.value = ''; fetchAllPositions(); } // Cüzdan sil function removeWallet(address) { walletAddresses = walletAddresses.filter(w => w !== address); saveWalletsToStorage(); updateWalletList(); fetchAllPositions(); } // Cüzdan listesini güncelle function updateWalletList() { const container = document.getElementById('walletList'); if (walletAddresses.length === 0) { container.innerHTML = '

Henüz cüzdan eklenmedi

'; return; } container.innerHTML = walletAddresses.map(address => `
${shortenAddress(address)} ✕
`).join(''); } // API: Kullanıcı pozisyonlarını çek async function fetchUserPositions(userAddress) { try { const response = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ type: 'clearinghouseState', user: userAddress }) }); if (!response.ok) throw new Error('API hatası'); const data = await response.json(); return data; } catch (error) { console.error('Pozisyon hatası:', error); return null; } } // API: Meta bilgilerini çek async function fetchMetadata() { try { const response = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ type: 'meta' }) }); if (!response.ok) throw new Error('API hatası'); return await response.json(); } catch (error) { console.error('Metadata hatası:', error); return null; } } // Tüm cüzdanların pozisyonlarını çek async function fetchAllPositions() { if (walletAddresses.length === 0) { allPositions = []; displayPositions(); updateStats(); return; } showLoading(true); allPositions = []; const metadata = await fetchMetadata(); const assetMap = {}; if (metadata && metadata.universe) { metadata.universe.forEach(asset => { assetMap[asset.name] = asset; }); } for (const address of walletAddresses) { const data = await fetchUserPositions(address); if (data && data.assetPositions) { for (const pos of data.assetPositions) { const position = pos.position; if (!position || position.szi === '0') continue; const coin = position.coin; const szi = parseFloat(position.szi); const side = szi > 0 ? 'Long' : 'Short'; const size = Math.abs(szi); const entryPx = parseFloat(position.entryPx || 0); const leverage = parseFloat(position.leverage?.value || 0); const unrealizedPnl = parseFloat(position.unrealizedPnl || 0); const returnOnEquity = parseFloat(position.returnOnEquity || 0); const liquidationPx = parseFloat(position.liquidationPx || 0); const cumFunding = parseFloat(position.cumFunding?.total || 0); let currentPrice = entryPx; if (size > 0 && unrealizedPnl !== 0) { if (side === 'Long') { currentPrice = entryPx + (unrealizedPnl / size); } else { currentPrice = entryPx - (unrealizedPnl / size); } } const value = size * currentPrice; const category = getWhaleCategory(value); allPositions.push({ wallet: address, coin: coin, side: side, leverage: leverage, size: size, entryPrice: entryPx, currentPrice: currentPrice, value: value, pnl: unrealizedPnl, pnlPercent: returnOnEquity * 100, fundingFee: cumFunding, liquidationPrice: liquidationPx, category: category }); } } await new Promise(resolve => setTimeout(resolve, 100)); } // Değere göre sırala (büyükten küçüğe) allPositions.sort((a, b) => b.value - a.value); showLoading(false); displayPositions(); updateStats(); } // Pozisyonları göster function displayPositions() { const tbody = document.getElementById('positionsBody'); // Filtrelenmiş pozisyonlar let filteredPositions = allPositions; if (currentFilter !== 'all') { filteredPositions = allPositions.filter(p => p.category === currentFilter); } if (filteredPositions.length === 0) { tbody.innerHTML = ` ${currentFilter === 'all' ? 'Açık pozisyon bulunamadı' : 'Bu kategoride pozisyon bulunamadı'} `; return; } tbody.innerHTML = filteredPositions.map(pos => { const categoryInfo = WHALE_CATEGORIES[pos.category]; const categoryClass = `whale-${pos.category}`; return ` ${shortenAddress(pos.wallet)} ${categoryInfo.emoji} ${categoryInfo.name} ${pos.coin} ${pos.side} ${pos.leverage.toFixed(1)}X Cross ${formatCurrency(pos.value)} ${formatNumber(pos.size)} ${pos.coin} $${formatNumber(pos.entryPrice)} ${pos.pnl >= 0 ? '+' : ''}${formatCurrency(pos.pnl)}
${pos.pnl >= 0 ? '+' : ''}${pos.pnlPercent.toFixed(2)}% ${pos.fundingFee >= 0 ? '+' : ''}${formatCurrency(pos.fundingFee)} $${formatNumber(pos.liquidationPrice)} `; }).join(''); } // İstatistikleri güncelle function updateStats() { const totalPositions = allPositions.length; const longPositions = allPositions.filter(p => p.side === 'Long').length; const shortPositions = allPositions.filter(p => p.side === 'Short').length; const totalValue = allPositions.reduce((sum, p) => sum + p.value, 0); const totalPnl = allPositions.reduce((sum, p) => sum + p.pnl, 0); const smallWhales = allPositions.filter(p => p.category === 'small').length; const regularWhales = allPositions.filter(p => p.category === 'whale').length; const tidalWhales = allPositions.filter(p => p.category === 'tidal').length; const megaWhales = allPositions.filter(p => p.category === 'mega').length; document.getElementById('totalPositions').textContent = totalPositions; document.getElementById('totalValue').textContent = formatCurrency(totalValue); const pnlElement = document.getElementById('totalPnl'); pnlElement.textContent = formatCurrency(totalPnl); pnlElement.className = totalPnl >= 0 ? 'positive' : 'negative'; document.getElementById('longPositions').textContent = longPositions; document.getElementById('shortPositions').textContent = shortPositions; document.getElementById('smallWhales').textContent = smallWhales; document.getElementById('regularWhales').textContent = regularWhales; document.getElementById('tidalWhales').textContent = tidalWhales; document.getElementById('megaWhales').textContent = megaWhales; } // Yardımcı fonksiyonlar function formatNumber(num) { if (Math.abs(num) >= 1000000) { return `${(num / 1000000).toFixed(2)}M`; } else if (Math.abs(num) >= 1000) { return `${(num / 1000).toFixed(2)}K`; } return num.toFixed(2); } function formatCurrency(num) { return `$${formatNumber(num)}`; } function shortenAddress(address) { if (!address) return 'N/A'; return `${address.slice(0, 6)}...${address.slice(-4)}`; } function showLoading(show) { document.getElementById('loadingIndicator').style.display = show ? 'block' : 'none'; } // Filtre butonları document.querySelectorAll('.filter-btn').forEach(btn => { btn.addEventListener('click', function() { document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active')); this.classList.add('active'); currentFilter = this.dataset.filter; displayPositions(); }); }); // Event listeners document.getElementById('addWalletBtn').addEventListener('click', addWallet); document.getElementById('refreshBtn').addEventListener('click', fetchAllPositions); document.getElementById('walletInput').addEventListener('keypress', (e) => { if (e.key === 'Enter') { addWallet(); } }); // Sayfa yüklendiğinde loadWalletsFromStorage(); if (walletAddresses.length > 0) { fetchAllPositions(); } console.log('🐋 Hyperliquid Whale Scanner hazır!'); console.log('Whale Categories:', WHALE_CATEGORIES);

// Hyperliquid API ile büyük pozisyon takibi async function fetchHyperliquidLargePositions() { try { // 1. Hyperliquid Info API'si - Açık pozisyonları çek const infoUrl = 'https://api.hyperliquid.xyz/info'; const requestBody = { type: 'clearinghouseState', user: '' // Boş bırakırsak tüm kullanıcıların özetini alırız }; const response = await fetch(infoUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(requestBody) }); const data = await response.json(); // 2. Büyük pozisyonları filtrele (örn: $500,000 üzeri) const largePositions = []; const MIN_POSITION_SIZE = 500000; // 500K USD if (data && data.assetPositions) { for (const user in data.assetPositions) { const positions = data.assetPositions[user]; positions.forEach(position => { // Pozisyon değerini hesapla const positionValue = parseFloat(position.positionValue); if (positionValue >= MIN_POSITION_SIZE) { largePositions.push({ user: user, coin: position.coin, size: parseFloat(position.szi), entryPx: parseFloat(position.entryPx), positionValue: positionValue, leverage: parseFloat(position.leverage), side: parseFloat(position.szi) > 0 ? 'LONG' : 'SHORT', unrealizedPnl: parseFloat(position.unrealizedPnl), liqPrice: parseFloat(position.liqPx) }); } }); } } // 3. Büyük pozisyonları değere göre sırala largePositions.sort((a, b) => b.positionValue - a.positionValue); return largePositions; } catch (error) { console.error('Hyperliquid API hatası:', error); return []; } } // 2. Alternatif: Sadece büyük ETH pozisyonlarını al async function fetchLargeETHPositions() { try { const infoUrl = 'https://api.hyperliquid.xyz/info'; const requestBody = { type: 'clearinghouseState', user: '' }; const response = await fetch(infoUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(requestBody) }); const data = await response.json(); const largeETHPositions = []; const MIN_ETH_VALUE = 1000000; // 1M USD if (data && data.assetPositions) { for (const user in data.assetPositions) { const positions = data.assetPositions[user]; positions.forEach(position => { // Sadece ETH pozisyonlarını filtrele if (position.coin === 'ETH' || position.coin === 'ETH-USD') { const positionValue = parseFloat(position.positionValue); if (positionValue >= MIN_ETH_VALUE) { largeETHPositions.push({ user: user.substring(0, 6) + '...' + user.substring(user.length - 4), size: parseFloat(position.szi), entryPrice: parseFloat(position.entryPx), positionValue: positionValue, leverage: parseFloat(position.leverage), side: parseFloat(position.szi) > 0 ? 'LONG' : 'SHORT', liqPrice: parseFloat(position.liqPx), timestamp: new Date().toISOString() }); } } }); } } // Değere göre sırala largeETHPositions.sort((a, b) => b.positionValue - a.positionValue); return largeETHPositions; } catch (error) { console.error('ETH pozisyonları hatası:', error); return []; } } // 3. Son büyük işlemleri çek async function fetchRecentLargeTrades() { try { const infoUrl = 'https://api.hyperliquid.xyz/info'; const requestBody = { type: 'userFills', user: '' // Tüm kullanıcılar için boş bırak }; const response = await fetch(infoUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(requestBody) }); const data = await response.json(); const largeTrades = []; const MIN_TRADE_SIZE = 250000; // 250K USD // Son 24 saat içindeki işlemleri filtrele const twentyFourHoursAgo = Date.now() - (24 * 60 * 60 * 1000); if (data && Array.isArray(data)) { data.forEach(trade => { const tradeTime = new Date(trade.time).getTime(); if (tradeTime > twentyFourHoursAgo) { // İşlem değerini hesapla const tradeValue = parseFloat(trade.sz) * parseFloat(trade.px); if (tradeValue >= MIN_TRADE_SIZE) { largeTrades.push({ user: trade.user.substring(0, 6) + '...' + trade.user.substring(trade.user.length - 4), coin: trade.coin, side: trade.side, size: parseFloat(trade.sz), price: parseFloat(trade.px), value: tradeValue, time: new Date(trade.time).toLocaleString('tr-TR'), hash: trade.hash || trade.tid }); } } }); } // Zamana göre sırala (en yeni ilk) largeTrades.sort((a, b) => new Date(b.time) - new Date(a.time)); return largeTrades.slice(0, 20); // Son 20 büyük işlem } catch (error) { console.error('Büyük işlemler hatası:', error); return []; } } // 4. Ana fonksiyon - Tüm verileri birleştir async function monitorHyperliquidWhales() { console.log('🐋 Hyperliquid Balina Takibi Başlıyor...\n'); // Tüm verileri paralel olarak çek const [largePositions, ethPositions, recentTrades] = await Promise.all([ fetchHyperliquidLargePositions(), fetchLargeETHPositions(), fetchRecentLargeTrades() ]); // Konsola rapor yazdır console.log('📊 BÜYÜK POZİSYONLAR ($500K+):'); console.log('==============================='); largePositions.slice(0, 10).forEach(pos => { console.log(`• ${pos.user.substring(0, 8)}... - ${pos.coin}: ${pos.side} $${(pos.positionValue/1000000).toFixed(2)}M (${pos.leverage}x) Liq: $${pos.liqPrice}`); }); console.log('\n🔥 BÜYÜK ETH POZİSYONLARI ($1M+):'); console.log('================================'); ethPositions.slice(0, 10).forEach(pos => { console.log(`• ${pos.user} - ${pos.side} ${pos.size} ETH ($${(pos.positionValue/1000000).toFixed(2)}M) ${pos.leverage}x Entry: $${pos.entryPrice} Liq: $${pos.liqPrice}`); }); console.log('\n⚡ SON BÜYÜK İŞLEMLER ($250K+):'); console.log('================================'); recentTrades.slice(0, 10).forEach(trade => { console.log(`• ${trade.time} - ${trade.user} ${trade.side} ${trade.size} ${trade.coin} @ $${trade.price} ($${(trade.value/1000000).toFixed(2)}M)`); }); console.log(`\n📈 Özet: Toplam ${largePositions.length} büyük pozisyon, ${ethPositions.length} büyük ETH pozisyonu, ${recentTrades.length} son büyük işlem`); return { largePositions, ethPositions, recentTrades, timestamp: new Date().toISOString() }; } // 5. WebSocket ile canlı takip (isteğe bağlı) function setupHyperliquidWebSocket() { const ws = new WebSocket('wss://api.hyperliquid.xyz/ws'); ws.onopen = () => { console.log('Hyperliquid WebSocket bağlantısı kuruldu'); // Subscribe to all fills (tüm işlemler) const subscribeMsg = { method: 'subscribe', subscription: { type: 'allMids' } }; ws.send(JSON.stringify(subscribeMsg)); }; ws.onmessage = (event) => { const data = JSON.parse(event.data); // Büyük işlemleri filtrele if (data.channel === 'fills') { const fill = data.data; const tradeValue = parseFloat(fill.sz) * parseFloat(fill.px); if (tradeValue > 100000) { // 100K USD üzeri console.log(`🔔 BÜYÜK İŞLEM: ${fill.side} ${fill.sz} ${fill.coin} @ $${fill.px} ($${(tradeValue/1000).toFixed(0)}K)`); // Bildirim göster showNotification(`${fill.coin} ${fill.side}`, `${fill.sz} @ $${fill.px} ($${(tradeValue/1000).toFixed(0)}K)`); } } }; ws.onerror = (error) => { console.error('WebSocket hatası:', error); }; ws.onclose = () => { console.log('WebSocket bağlantısı kapandı, 5 saniye sonra yeniden bağlanılıyor...'); setTimeout(setupHyperliquidWebSocket, 5000); }; return ws; } // 6.

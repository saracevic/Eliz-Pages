// Global değişkenler let allTrades = []; let autoRefreshInterval = null; let isAutoRefresh = false; // Hyperliquid API endpoint const API_URL = 'https://api.hyperliquid.xyz/info'; // Top 50 coin listesi (Hyperliquid'de popüler olanlar) const TOP_COINS = [ 'BTC', 'ETH', 'SOL', 'ARB', 'MATIC', 'AVAX', 'DOGE', 'OP', 'ATOM', 'SUI', 'SEI', 'APT', 'TIA', 'WLD', 'INJ', 'ORDI', 'FTM', 'NEAR', 'STX', 'MEME', 'BONK', 'PENDLE', 'WIF', 'RNDR', 'IMX', 'BLUR', 'LDO', 'RUNE', 'JTO', 'STRK', 'PYTH', 'JUP', 'DYDX', 'GMX', 'GRT', 'FET', 'AAVE', 'MKR', 'COMP', 'UNI', 'SUSHI', 'CRV', 'SNX', 'YFI', 'BAL', 'LRC', '1INCH', 'ZRX', 'LINK', 'DOT' ]; // Whale wallet adresleri (bilinen büyük cüzdanlar) const KNOWN_WHALES = new Set(); // Yardımcı fonksiyonlar function formatNumber(num) { if (num >= 1000000) { return `${(num / 1000000).toFixed(2)}M`; } else if (num >= 1000) { return `${(num / 1000).toFixed(2)}K`; } return num.toFixed(2); } function formatCurrency(num) { return `$${formatNumber(num)}`; } function formatTime(timestamp) { const date = new Date(timestamp); const now = new Date(); const diff = Math.floor((now - date) / 1000); if (diff < 60) return `${diff} saniye önce`; if (diff < 3600) return `${Math.floor(diff / 60)} dakika önce`; if (diff < 86400) return `${Math.floor(diff / 3600)} saat önce`; return date.toLocaleString('tr-TR'); } function shortenAddress(address) { if (!address) return 'N/A'; return `${address.slice(0, 6)}...${address.slice(-4)}`; } // API fonksiyonları async function fetchMetadata() { try { const response = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ type: 'meta' }) }); if (!response.ok) throw new Error('API hatası'); return await response.json(); } catch (error) { console.error('Metadata hatası:', error); return null; } } async function fetchUserFills(userAddress) { try { const response = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ type: 'userFills', user: userAddress }) }); if (!response.ok) throw new Error('API hatası'); return await response.json(); } catch (error) { console.error('User fills hatası:', error); return []; } } async function fetchRecentTrades(coin) { try { const response = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ type: 'recentTrades', coin: coin }) }); if (!response.ok) throw new Error('API hatası'); return await response.json(); } catch (error) { console.error(`${coin} işlem hatası:`, error); return []; } } async function scanForWhales() { showLoading(true); const metadata = await fetchMetadata(); if (!metadata || !metadata.universe) { showLoading(false); showError('Metadata yüklenemedi'); return; } const trades = []; const minSize = parseFloat(document.getElementById('minSize').value) || 50000; const coinFilter = document.getElementById('coinFilter').value.toUpperCase().trim(); // Filtrelenmiş coin listesi let coinsToScan = TOP_COINS; if (coinFilter) { coinsToScan = TOP_COINS.filter(c => c.includes(coinFilter)); } // Her coin için son işlemleri çek for (const coin of coinsToScan.slice(0, 20)) { // Rate limit için ilk 20 coin const recentTrades = await fetchRecentTrades(coin); if (Array.isArray(recentTrades)) { for (const trade of recentTrades) { const tradeSize = parseFloat(trade.sz || 0); const tradePrice = parseFloat(trade.px || 0); const tradeValue = tradeSize * tradePrice; // Büyük işlemleri filtrele if (tradeValue >= minSize) { trades.push({ coin: coin, side: trade.side === 'B' ? 'BUY' : 'SELL', size: tradeSize, price: tradePrice, value: tradeValue, time: trade.time, hash: trade.tid || 'N/A' }); } } } // Rate limit için kısa bekleme await new Promise(resolve => setTimeout(resolve, 100)); } allTrades = trades.sort((a, b) => b.time - a.time); showLoading(false); displayTrades(); updateStats(); } // UI fonksiyonları function showLoading(show) { document.getElementById('loadingIndicator').style.display = show ? 'block' : 'none'; } function showError(message) { const container = document.getElementById('tradesContainer'); container.innerHTML = `
${message}
`; } function displayTrades() { const container = document.getElementById('tradesContainer'); if (allTrades.length === 0) { container.innerHTML = '
Balina işlemi bulunamadı
'; return; } container.innerHTML = allTrades.map(trade => `
${trade.coin} ${trade.side} ${formatTime(trade.time)}
Miktar ${formatNumber(trade.size)} ${trade.coin}
Fiyat ${formatCurrency(trade.price)}
Toplam Değer ${formatCurrency(trade.value)}
İşlem ID ${trade.hash}
`).join(''); } function updateStats() { const totalTrades = allTrades.length; const buyTrades = allTrades.filter(t => t.side === 'BUY').length; const sellTrades = allTrades.filter(t => t.side === 'SELL').length; const totalVolume = allTrades.reduce((sum, t) => sum + t.value, 0); document.getElementById('totalTrades').textContent = totalTrades; document.getElementById('totalVolume').textContent = formatCurrency(totalVolume); document.getElementById('buyTrades').textContent = buyTrades; document.getElementById('sellTrades').textContent = sellTrades; } function toggleAutoRefresh() { isAutoRefresh = !isAutoRefresh; const btn = document.getElementById('autoRefreshBtn'); if (isAutoRefresh) { btn.textContent = '⏹️ Otomatik Durdur'; btn.style.background = 'linear-gradient(135deg, #ef4444 0%, #dc2626 100%)'; autoRefreshInterval = setInterval(scanForWhales, 30000); // 30 saniyede bir scanForWhales(); } else { btn.textContent = '⏱️ Otomatik Yenileme'; btn.style.background = 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)'; if (autoRefreshInterval) { clearInterval(autoRefreshInterval); autoRefreshInterval = null; } } } // Event listeners document.getElementById('refreshBtn').addEventListener('click', scanForWhales); document.getElementById('autoRefreshBtn').addEventListener('click', toggleAutoRefresh); document.getElementById('coinFilter').addEventListener('keypress', (e) => { if (e.key === 'Enter') { scanForWhales(); } }); document.getElementById('minSize').addEventListener('keypress', (e) => { if (e.key === 'Enter') { scanForWhales(); } }); // Sayfa yüklendiğinde console.log('🐋 Hyperliquid Whale Scanner hazır!'); console.log('Top coins:', TOP_COINS);
